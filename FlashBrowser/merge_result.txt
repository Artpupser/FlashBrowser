using FlashBrowser.Core.Pipelines;
using FlashBrowser.Core.Pipelines.Context;

namespace FlashBrowser;

public sealed class App(StartupPipeline startupPipeline, CorePipeline corePipeline) : IApp, IDisposable {
   private CancellationTokenSource? _linkedCts;
   private IPipelineContext? _startupContext;

   public async Task Run(CancellationToken cancellationToken) {
      _linkedCts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
      var contextOption = await startupPipeline.RunAsync(cancellationToken);
      if (contextOption.Out(out _startupContext))
         await corePipeline.Execute(_startupContext, cancellationToken);
   }

   public async Task Stop() {
      if (_linkedCts != null)
         await _linkedCts.CancelAsync();
   }

   public void Dispose() {
      _linkedCts?.Dispose();
   }
}

using FlashBrowser.Core.Pipelines.Context;
using FlashBrowser.Extensions;
using FlashBrowser.UI.Builders;
using FlashBrowser.UI.Styles;
using FlashBrowser.UI.Values;

using PupaLib.Core;

namespace FlashBrowser.Core.BrowserStrategy;

public sealed class ClearBrowserCookiesStrategy(IUiBuilder ui) : ICoreStrategy {
   public string Title => "Clear cookies";

   public async Task<Option> Execute(IPipelineContext context, CancellationToken cancellationToken = default) {
      while (cancellationToken.IsCancellationRequested == false) {
         var openedPages = context.BrowserService.OpenedPages.Where(x => x.IsClosed == false).ToArray();
         string[] openedPagesNames = [.. openedPages.Select(x => new Uri(x.Url).Host), "(danger) all"];
         var siteRef = new ValueRef<(int index, string name)>();
         ui.WithChain(chain => chain
            .WithChoice(choice => choice
               .WithChoices(openedPagesNames)
               .WithPrompt("Select a site")
               .Ref(siteRef)));

         if (await ui.RunAndClear(cancellationToken) && siteRef.Get().index > -1) {
            var all = siteRef.Get().index == openedPagesNames.Length - 1;
            var verifyMessage = new ValueRef<string>(all
               ? $"Do you want clear all cookies, from browser {context.BrowserService.ActiveBrowser.GetLowerName()}"
               : $"Do you want clear cookies, host '{siteRef.Get().name}'");
            var verifyRef = new ValueRef<bool>();
            ui.WithChain(chain => chain
               .WithVerify(verify => verify
                  .WithPrompt(verifyMessage)
                  .Ref(verifyRef)));
            if (await ui.RunAndClear(cancellationToken) && verifyRef.Get()) {
               var pages = all ? openedPages : [openedPages[siteRef.Get().index]];
               foreach (var page in pages) {
                  var cookies = await page.GetCookiesAsync();
                  foreach (var cookie in cookies)
                     await ui.Message($"Find cookie:  {cookie.Domain} {cookie.Name} {cookie.Expires}",
                        cancellationToken);
                  await page.DeleteCookieAsync(cookies);
                  await ui.Message($"Removing all cookies, host '{siteRef.Get().name}'", cancellationToken);
               }
            }

            continue;
         }

         break;
      }

      return Option.Fail();
   }
}

using FlashBrowser.Core.Killers;
using FlashBrowser.Core.Pipelines.Context;
using FlashBrowser.Extensions;
using FlashBrowser.Services.Browser;
using FlashBrowser.UI.Builders;
using FlashBrowser.UI.Styles;

using PupaLib.Core;

namespace FlashBrowser.Core.BrowserStrategy;

public sealed class KillBrowserStrategy(IUiBuilder ui, BrowserService browserService, IProcessKiller processKiller)
   : ICoreStrategy {
   public string Title => $"Kill all browser <{browserService.ActiveBrowser}> process";

   public async Task<Option> Execute(IPipelineContext context, CancellationToken cancellationToken = default) {
      foreach (var killResult in processKiller.KillByPrefix(browserService.ActiveBrowser.GetLowerName())) {
         if (killResult.IsSuccess) {
            await ui.Message($"Kill process '{killResult.Name}', {killResult.Id}", cancellationToken);
            continue;
         }

         await ui.Message($"Ignore killing process with name '{killResult.Name}', {killResult.Error!.Message}",
            cancellationToken);
      }

      return Option.Ok();
   }
}

public sealed class LaunchBrowserStrategy(IUiBuilder ui, BrowserService browserService) : ICoreStrategy {
   public string Title => $"Launch selected browser <{browserService.ActiveBrowser}>";
   // ui -> uiBuilder

   public async Task<Option> Execute(IPipelineContext context, CancellationToken cancellationToken = default) {
      if (context.BrowserService.Browser == null) {
         var activeBrowser = context.BrowserService.ActiveBrowser;
         try {
            await context.BrowserService.LaunchBrowser();
            await ui.Message(
               $"Launch browser <{activeBrowser}>, use '{context.BrowserService.BrowserApps[activeBrowser]}'",

            await ui.Message($"Failed launch browser <{activeBrowser}> exception: {e.Message}", cancellationToken);
            return Option.Fail();
         }
      }

      return Option.Fail();
   }
}

using System.Diagnostics;

using FlashBrowser.Core.Factory;
using FlashBrowser.Core.Pipelines.Context;
using FlashBrowser.Core.ProfileStrategy;
using FlashBrowser.Core.ProfileStrategy.Context;
using FlashBrowser.Services.Browser;
using FlashBrowser.UI.Builders;
using FlashBrowser.UI.Styles;

using PupaLib.Core;

namespace FlashBrowser.Core.BrowserStrategy;

public sealed class OpenLaunchProfileInBrowserStrategy(
   IUiBuilder ui,
   IStrategyFactory<IProfileStrategy<LaunchProfile>> launchProfilesFactory) : ICoreStrategy {
   public string Title => "Open profile";

   public async Task<Option> Execute(IPipelineContext context, CancellationToken cancellationToken = default) {
      var strategies = launchProfilesFactory.Create();
      var selectStrategy = strategies.First(x => x is SelectProfileStrategy);
      var profileContext = new LaunchProfileContext(context.DataService.ApplicationData!.Profiles);
      while (cancellationToken.IsCancellationRequested == false && profileContext.SelectedProfile == null)
         if (await selectStrategy.Execute(profileContext, cancellationToken) == false)
            return Option.Fail();

      var selectedProfile = profileContext.SelectedProfile!;
      if (context.BrowserService.Browser != null) {
         for (var i = 0; i < selectedProfile.Links.Count; i++) {
            var link = selectedProfile.Links[i];
            try {
               await context.BrowserService.NewPageTab(link);
               await ui.Message($"Open link, '{link}'", cancellationToken);
            } catch (Exception e) {
               await ui.Message($"Failed open link, '{link}' message, {e.Message}", cancellationToken);
            }
         }

         for (var i = 0; i < selectedProfile.AppProfiles.Count; i++) {
            var appProfile = selectedProfile.AppProfiles[i];
            var executePath = appProfile.GetPathOrEmpty(context.OperatingSystemService.CurrentOperatingSystem);
            try {
               if (string.IsNullOrWhiteSpace(executePath))
                  throw new Exception(
                     $"execute path for '{context.OperatingSystemService.CurrentOperatingSystem}' operating system undefined");
               Process.Start(new ProcessStartInfo {
                  UseShellExecute = true,
                  FileName = executePath,
                  WindowStyle = ProcessWindowStyle.Maximized
               });
               await ui.Message($"Open application, '{executePath}'", cancellationToken);
            } catch (Exception e) {
               await ui.Message($"Failed open application, '{executePath}' message, {e.Message}", cancellationToken);
            }
         }

         return Option.Ok();
      }

      await ui.Message("Browser not run, launch any browser and retry", cancellationToken);
      return Option.Fail();
   }
}

﻿using FlashBrowser.Core.Pipelines.Context;
using FlashBrowser.Services.Browser;
using FlashBrowser.UI.Builders;
using FlashBrowser.UI.Styles;
using FlashBrowser.UI.Values;

using PupaLib.Core;

using PuppeteerSharp;

namespace FlashBrowser.Core.BrowserStrategy;

public sealed class SelectBrowserStartegy(IUiBuilder ui) : ICoreStrategy {
   public string Title => "Select browser for work";

   public async Task<Option> Execute(IPipelineContext context, CancellationToken cancellationToken = default) {
      var loadedBrowsers = context.BrowserService.BrowserApps
         .Select(x => (e: x.Key, name: context.BrowserService.GetNameFromEnum(x.Key))).ToArray();
      var loadedBrowserNames = loadedBrowsers.Select(x => x.name).ToArray();

      if (loadedBrowsers.Length == 1) {
         SelectBrowser(context.BrowserService, loadedBrowsers.First().e);
         return Option.Ok();
      }

      var selectBrowseRef = new ValueRef<(int index, string name)>();

      while (cancellationToken.IsCancellationRequested == false) {
         ui.WithChain(chain => chain
            .WithChoice(choice => choice
               .WithChoices(loadedBrowserNames)
               .WithPrompt("Select browser for work")
               .Ref(selectBrowseRef)));
         if (await ui.RunAndClear(cancellationToken) == false)
            break;

         SelectBrowser(context.BrowserService, loadedBrowsers[selectBrowseRef.Get().index].e);
         return Option.Ok();
      }

      return Option.Fail();
   }

   private void SelectBrowser(BrowserService browserService, SupportedBrowser browser) {
      browserService.SetActiveBrowser(browser);
      ui.Message($"Active browser {browser} selected");
   }
}

using FlashBrowser.Core.BrowserStrategy;
using FlashBrowser.Services.Browser;

namespace FlashBrowser.Core.Factory;

public class BrowserCommandsFactory(BrowserService browserService, IEnumerable<ICoreStrategy> strategies)
   : IStrategyFactory<ICoreStrategy> {
   public IEnumerable<ICoreStrategy> Create() {
      return strategies
         .Where(x => !(x is ClearBrowserCookiesStrategy && browserService.OpenedPages.Count <= 0))
         .Where(x => !(x is CloseAllBrowserTabsStrategy && browserService.OpenedPages.Count <= 0))
         .Where(x => !(x is OpenLaunchProfileInBrowserStrategy && browserService.Browser == null))
         .Where(x => !(x is LaunchBrowserStrategy && browserService.Browser != null))
         .Where(x => !(x is UnlaunchBrowserStrategy && browserService.Browser == null))
         .Where(x => !(x is KillBrowserStrategy && browserService.Browser != null));
   }
}

using FlashBrowser.Core.Os;

using PupaLib.FileIO;

namespace FlashBrowser.Core.FileSearch;

public sealed class FindApplicationsOfExtensionStrategy(DefaultOsProgramFolderStrategy defaultOsProgramFolderStrategy) {
   public string Title => "Find applications of extension";

   public IEnumerable<VirtualFile> Execute(string extension, CancellationToken cancellationToken) {
      if (cancellationToken.IsCancellationRequested)
         yield break;
      if (!defaultOsProgramFolderStrategy.Execute().Out(out var defaultApplicationsFolder))
         yield break;
      var dir = new DirectoryInfo(defaultApplicationsFolder.MyPath);
      var enumerateOption = new EnumerationOptions {
         MatchCasing = MatchCasing.CaseInsensitive,
         IgnoreInaccessible = true
      };
      foreach (var file in dir.EnumerateFiles($"*.{extension}", enumerateOption))
         if (file.Extension == $".{extension}" && VirtualFile.GetFile(file.FullName).Out(out var virtualFile))
            yield return virtualFile;
   }
}

   public async Task<Option<VirtualFolder>> Execute(IPipelineContext context, CancellationToken cancellationToken) {

      var history = localConfig.LastBrowserFoldersHistory;
      if (VirtualFolder.GetFolder(path).Out(out var browserFolder)) return Option<VirtualFolder>.Ok(browserFolder);

      await ui.Message($"Folder with path '{path}' undefined, in any drive", cancellationToken);

            .WithVerify(verify => verify
               .WithPrompt($"Remove path from history? '{path}'")
               .Ref(acceptRef)));
      await ui.RunAndClear(cancellationToken);
      if (acceptRef.Get()) {
         history.Remove(browserFolder.MyPath);
         localConfig.LastBrowserFoldersHistory = history;
      }

      return Option<VirtualFolder>.Fail();
   }
}

   : IFolderSearchStrategy {

         if (verifyValue.Get() == false) {
            var changedFolderNameOption = await typeFolderNameStrategy.Execute(context, cancellationToken);

         await ui.Message(
            new ConstValue<string>($"Folder '{browserFolderName.Get()}' undefined, in drive <{usableDrive.Name}>"),
            cancellationToken);
      }

      return Option<VirtualFolder>.Fail();
   }
}

using System.Diagnostics;

namespace FlashBrowser.Core.Killers;

public sealed class ProcessKiller : IProcessKiller {
   public IEnumerable<KillResult> KillByPrefix(string prefix) {
      foreach (var el in Process.GetProcesses()) {
         if (el.ProcessName.StartsWith(prefix) == false)
            continue;
         yield return Kill(el);
         el.Dispose();
      }
   }

   public IEnumerable<KillResult> KillByContains(string name) {
      foreach (var el in Process.GetProcesses()) {
         if (el.ProcessName.Contains(name) == false)
            continue;
         yield return Kill(el);
         el.Dispose();
      }
   }

   public KillResult Kill(Process process) {
      try {
         process.Kill();
         return new KillResult(process.ProcessName, process.Id);
      } catch (Exception e) {
         return new KillResult(process.ProcessName, process.Id, e);
      }
   }
}

using FlashBrowser.Services.OperationSystem;

using PupaLib.Core;

using PuppeteerSharp;

namespace FlashBrowser.Core.Os;

public sealed class GetOsSupportedBrowsersStrategy(OperatingSystemService operatingSystemService) {
   public Option<IEnumerable<SupportedBrowser>> Execute() {
      return Option<IEnumerable<SupportedBrowser>>.Ok(Get());
   }

   private IEnumerable<SupportedBrowser> Get() {
      yield return SupportedBrowser.Firefox;

﻿using FlashBrowser.Core.Factory;
using FlashBrowser.Core.Pipelines.Context;
using FlashBrowser.UI.Builders;
using FlashBrowser.UI.Styles;
using FlashBrowser.UI.Values;

using PupaLib.Core;

namespace FlashBrowser.Core.Pipelines;

public sealed class CorePipeline(
   IUiBuilder ui,
   IStrategyFactory<ICoreStrategy> commandsFactory) {
   public async Task<Option> Execute(IPipelineContext context, CancellationToken cancellationToken = default) {
      const byte maxExitAttempts = 3;
      var exitAttempts = maxExitAttempts;
      while (cancellationToken.IsCancellationRequested == false) {
         var strategiesArr = commandsFactory.Create().ToArray();
         var strategiesNames = strategiesArr.Select(x => x.Title).ToArray();
         var actionRef = new ValueRef<(int index, string name)>();
         ui.WithChain(chain => chain
            .WithChoice(choice => choice
               .WithChoices(strategiesNames)
               .WithPrompt("Select action")
               .Ref(actionRef)));
         if (await ui.RunAndClear(cancellationToken) && actionRef.Get().index > -1) {
            var strategy = strategiesArr[actionRef.Get().index];
            exitAttempts = maxExitAttempts;
            await strategy.Execute(context, cancellationToken);
         } else {
            exitAttempts -= 1;
            if (exitAttempts == 0)
               return Option.Ok();
            await ui.Message($"If you want exit, press ESC <{exitAttempts}/{maxExitAttempts}> once", cancellationToken);
         }
      }

      return Option.Fail();
   }
}

             getOsSupportedBrowsersStrategy.Execute().Out(out var supportedBrowsers)) {
         var osName = currentOs.GetLowerName();
         await ui.Message($"Set application path for '{osName}' os", cancellationToken);
         var arr = ApplySupportedBrowsers(context.BrowserService, supportedBrowsers, cancellationToken)
            .ToArray();
         if (arr.Any())
            return Option.Ok();
      }

      await ui.Message("You not have a web browser, download any supported browser correct work this application",
         cancellationToken);
      return Option.Fail();
   }

   private IEnumerable<Option> ApplySupportedBrowsers(BrowserService browserService,
      IEnumerable<SupportedBrowser> supportedBrowsers, [EnumeratorCancellation] CancellationToken cancellationToken) {
      foreach (var supportedBrowser in supportedBrowsers) {
         var collection = findApplicationsOfNameStrategy.Execute(supportedBrowser.GetLowerName(), cancellationToken)
            .ToArray();
         if (collection.Any()) yield return ApplyApp(browserService, supportedBrowser, collection.First().MyPath);
      }
   }

   private Option ApplyApp(BrowserService browserService, SupportedBrowser browser, string path) {
      if (browserService.SetApplication(browser, path)) {
         ui.Message($"Apply new path <{path}> for browser '{browserService.GetNameFromEnum(browser)}'");
         return Option.Ok();
      }

      ui.Message($"Browser '{browserService.GetNameFromEnum(browser)}' undefined with path <{path}> ");
      return Option.Fail();
   }
}

         ui

               .WithInput<string>(input => input
                  .WithPrompt("Repeat a new password")
                  .WithRange(8, 20)
                  .Ref(repeatNewPasswordRef))
            );
         await ui.RunAndClear(cancellationToken);
         if (newPasswordRef.Get() == repeatNewPasswordRef.Get()) {

            localConfig.Password = MyMath.HashSha256(newPasswordRef.Get());
            await context.DataService.Save(cancellationToken);
         }
      }

      const byte maxAttempts = 3;
      var attempts = maxAttempts;
      while (cancellationToken.IsCancellationRequested == false && localConfig.IsFirst == false &&
             isNewPassword == false) {
         await ui.Message("Hello user, it's FlashBrowser app", cancellationToken);
         var passwordRef = new ValueRef<string>();

            .WithChain(chain => chain
               .WithInput<string>(input => input
                  .WithPrompt("Type a password")
                  .WithRange(8, 20)
                  .WithSecret('*')
                  .Ref(passwordRef)));
         await ui.RunAndClear(cancellationToken);
         if (MyMath.HashSha256(passwordRef.Get()) != localConfig.Password) {
            attempts -= 1;
            if (attempts < 1) return Option.Fail();
            await ui.Message($"Wrong password, attempts {attempts}/{maxAttempts}", cancellationToken);
            continue;
         }

         break;
      }

using FlashBrowser.Core.FolderSearch;
using FlashBrowser.Core.Pipelines.Context;
using FlashBrowser.UI.Builders;
using FlashBrowser.UI.Values;

using PupaLib.Core;

namespace FlashBrowser.Core.Startup;

public sealed class SelectBrowserFolderStrategy(
   IUiBuilder ui,
   IEnumerable<IFolderSearchStrategy> folderSearchStrategies) : IStartupStrategy {
   public string Title => "Select browser folder";

   public async Task<Option> Execute(IPipelineContext context, CancellationToken cancellationToken) {
      var localConfig = context.DataService.ApplicationData!;
      var searchStrategyList = folderSearchStrategies.ToList();
      var success = false;
      while (cancellationToken.IsCancellationRequested == false) {
         var choices = searchStrategyList.Select(x => x.Title).Concat(localConfig.LastBrowserFoldersHistory).ToArray();
         var searchChoiceRef = new ValueRef<(int index, string name)>();
         ui.WithChain(chain => chain
            .WithChoice(choice => choice
               .WithChoices(choices)
               .WithPrompt("Search variant")
               .Ref(searchChoiceRef)));

         await ui.RunAndClear(cancellationToken);
         var searchIndex = searchChoiceRef.Get().index;
         var strategy = searchIndex < searchStrategyList.Count
            ? searchStrategyList[searchIndex]
            : new HistoryStrategy(localConfig.LastBrowserFoldersHistory[searchIndex - searchStrategyList.Count], ui);

         var searchOption = await strategy.Execute(context, cancellationToken);
         if (searchOption.Out(out var folder)) {
            success = true;
            context.BrowserService.BrowserEnvironmentFolder = folder;
            Remember(context);
         }

         await context.DataService.Save(cancellationToken);
         if (success)
            return Option.Ok();
      }

      return Option.Fail();
   }

   private static void Remember(IPipelineContext context) {
      var folder = context.BrowserService.BrowserEnvironmentFolder;
      var localConfig = context.DataService.ApplicationData!;
      var h = localConfig.LastBrowserFoldersHistory;
      h.Remove(folder.MyPath);
      h.Add(folder.MyPath);
      if (h.Count > 20)
         h.RemoveAt(0);
      localConfig.LastBrowserFoldersHistory = h;
   }
}

   public IReadOnlyDictionary<SupportedBrowser, VirtualFolder> BrowserFolders => _browserFolders;
   public IReadOnlyDictionary<SupportedBrowser, string> BrowserApps => _browserApps;
   public IReadOnlyList<IPage> OpenedPages => _openedPages;
   public IBrowser? Browser { get; private set; }
   public VirtualFolder BrowserEnvironmentFolder { get; set; } = null!;
   public SupportedBrowser ActiveBrowser { get; private set; }

   public DataService DataService { get; }

   public async Task DeactivateBrowser() {
      if (Browser == null)
         return;
      await CloseAllPages();
      await Browser.CloseAsync();
      Browser.Process.Kill();
      Browser.Process.Dispose();
      await Browser.DisposeAsync();
      Browser = null;
   }

   public async Task LaunchBrowser() {
      Browser = await Puppeteer.LaunchAsync(_browserLaunchOptionsFactory
         .Create(ActiveBrowser)
         .Execute(BrowserApps[ActiveBrowser], BrowserFolders[ActiveBrowser].MyPath));
   }


   public async Task CloseAllPages(Action<IPage>? closeCallback = null) {
      if (Browser == null)
         return;

      foreach (var page in _openedPages) {
         await page.CloseAsync();
         closeCallback?.Invoke(page);
         await page.DisposeAsync();
      }

      _openedPages.Clear();
   }

   public string GetNameFromEnum(SupportedBrowser supportedBrowser) {
      return supportedBrowser.ToString().ToLower();
   }

   public void SetActiveBrowser(SupportedBrowser supportedBrowser) {
      ActiveBrowser = supportedBrowser;
   }

   public void SetFolderProfile(SupportedBrowser supportedBrowser, VirtualFolder folder) {
      _browserFolders[supportedBrowser] = folder;
   }

   public Option SetApplication(SupportedBrowser supportedBrowser, string path) {
      if (File.Exists(path)) {
         _browserApps[supportedBrowser] = path;
         return Option.Ok();
      }

      return Option.Fail();
   }

   public async Task NewPageTab(string url) {
      if (Browser == null)
         return;
      try {
         var page = await Browser.NewPageAsync(new CreatePageOptions {
            Type = CreatePageType.Tab,
            WindowBounds = new WindowBounds {
               Width = 1920,
               Height = 1090,
               WindowState = WindowState.Fullscreen
            }
         });
         _ = page.GoToAsync(url);
         _ = page.ResizeAsync(1920, 1080);
         _openedPages.Add(page);
      } catch (Exception) {
         // ignored
      }
   }

   public void ForEachSupportedBrowsers(Action<SupportedBrowser, int> callback) {
      foreach (var item in _browserApps.Keys) {
         if (OperatingSystem.IsWindows() && item == SupportedBrowser.Chromium)
            continue;
         callback(item, (int)item);
      }
   }
}

namespace FlashBrowser.Services.Input;

public sealed class ConsoleInputService(IBindPressed bindPressed) : IKeyboardInputService {
   private readonly Eventer<ConsoleKeyInfo> _keyPressed = new();

   public Task Init(CancellationToken cancellationToken) {
      Console.CancelKeyPress += (_, _) => { bindPressed.OnCancel(); };
      _ = Task.Run(() => {
         while (cancellationToken.IsCancellationRequested == false) {
            var key = Console.ReadKey(true);
            _keyPressed.Invoke(this, key);
            if ((key.Modifiers & ConsoleModifiers.Control) != 0 && key.Key == ConsoleKey.Z) bindPressed.OnCtrlZ();
         }
      }, cancellationToken);
      return Task.CompletedTask;
   }

   public async Task<KeyboardKey> GetKeyboardKey() {
      var result = await GetConsoleKey();
      return KeyboardKey.FromConsole(result);
   }

   public async Task<ConsoleKeyInfo> GetConsoleKey() {
      var tcs = new TaskCompletionSource<ConsoleKeyInfo>();
      var handler = _keyPressed.Subscribe((sender, info) => { tcs.SetResult(info); });
      var result = await tcs.Task;
      _keyPressed.Unsubscribe(handler);
      return result;
   }
}

public sealed class UiBuilder : IUiBuilder {

   private readonly ServiceProvider _serviceProvider;

   public UiBuilder(IUiEngine engine, IKeyboardInputService keyboardInputService) {
      _chains = [];
      KeyboardInputService = keyboardInputService;
      ServiceCollection serviceCollection = [];
      serviceCollection
         .AddSingleton<IKeyboardInputService>(x => keyboardInputService)
         .AddSingleton<IUiEngine>(x => engine);
      _serviceProvider = serviceCollection.BuildServiceProvider();
   }

   public IKeyboardInputService KeyboardInputService { get; }

   public IUiBuilder WithChain(Action<IUiBuilderChain> chainBuilder) {
      var chain = new UiBuilderChain(_serviceProvider);
      chainBuilder.Invoke(chain);
      _chains.Enqueue(chain);
      return this;
   }

   public async Task<Option> Run(CancellationToken cancellationToken) {
      while (_chains.Count > 0) {
         var chain = _chains.Dequeue();
         var result = await chain.Run(cancellationToken);
         if (result.Success)
            continue;
         return Option.Fail();
      }

      return Option.Ok();
   }

   public async Task<Option> RunAndClear(CancellationToken cancellationToken) {
      var result = await Run(cancellationToken);
      Clear();
      return result;
   }

   public IUiBuilder Clear() {
      _chains.Clear();
      return this;
   }
}

using FlashBrowser.Services.Input;
using FlashBrowser.UI.Elements.Base;
using FlashBrowser.UI.Values;
using FlashBrowser.Utils;

using PupaLib.Core;

namespace FlashBrowser.UI.Elements.Terminal;

public sealed class UiTerminalChoice(IKeyboardInputService keyboardInputService) : UiElement, IUiChoice {
   private readonly List<IValue<string>> _itemsValues = [];
   private readonly StringBuilder _sb = new();
   private readonly string EmptyLine = new(' ', Console.WindowWidth);
   private IValue<Color> _noSelectBg = new ConstValue<Color>(Color.Black);
   private IValue<Color> _noSelectFg = new ConstValue<Color>(Color.White);
   private IValue<string> _promptValue = new ConstValue<string>(string.Empty);
   private IValue<Color> _selectBg = new ConstValue<Color>(Color.White);
   private IValue<Color> _selectFg = new ConstValue<Color>(Color.Black);
   private ValueRef<(int, string)>? _valueRef;

   public override async Task<Option> Run(CancellationToken cancellationToken) {
      while (cancellationToken.IsCancellationRequested == false) {
         var result = await Invoke(cancellationToken);
         Console.CursorVisible = true;
         if (result.Success == false)
            break;
         _valueRef?.Set(result.Content);
         return Option.Ok();
      }

      return Option.Fail();
   }

   public IUiChoice WithPrompt(IValue<string> value) {
      _promptValue = value;
      return this;
   }

   public IUiChoice WithChoice(IValue<string> value) {
      _itemsValues.Add(value);
      return this;
   }

   public IUiChoice WithChoices(params IValue<string>[] value) {
      _itemsValues.AddRange(value);
      return this;
   }

   public IUiChoice WithSelectFgColor(IValue<Color> color) {
      _selectFg = color;
      return this;
   }

   public IUiChoice WithSelectBgColor(IValue<Color> color) {
      _selectBg = color;
      return this;
   }

   public IUiChoice WithNoSelectFgColor(IValue<Color> color) {
      _noSelectFg = color;
      return this;
   }

   public IUiChoice WithNoSelectBgColor(IValue<Color> color) {
      _noSelectBg = color;
      return this;
   }

   public IUiChoice Ref(in ValueRef<(int index, string choice)> valueRef) {
      _valueRef = valueRef;
      return this;
   }

   private void ClearLines(int amount) {
      for (var i = 0; i <= amount + 1; i++)
         _sb.AppendLine(EmptyLine);
      Console.Write(_sb.ToString());
      _sb.Clear();
   }

   private async Task<Option<(int, string)>> Invoke(CancellationToken cancellationToken) {
      var index = 0;
      ClearLines(_itemsValues.Count);
      Console.CursorVisible = false;
      var startCursor = (left: 0, top: Console.CursorTop);
      while (cancellationToken.IsCancellationRequested == false) {
         Console.CursorTop = startCursor.top - _itemsValues.Count - 1;
         _sb.AppendLine(Color.ColorizeFgBgX1B($"<< {_promptValue.Get()} >>", _noSelectFg.Get(), _noSelectBg.Get()));
         for (var i = 0; i < _itemsValues.Count; i++) {
            var item = _itemsValues[i];
            var value = item.Get();
            if (i == index) {
               _sb.AppendLine(Color.ColorizeFgBgX1B($"-> {value}", _selectFg.Get(), _selectBg.Get()));
               continue;
            }

            _sb.AppendLine(Color.ColorizeFgBgX1B($"   {value}", _noSelectFg.Get(), _noSelectBg.Get()));
         }

         Console.Write(_sb.ToString());
         _sb.Clear();
         //Console.SetCursorPosition(0, startCursor.top);
         var key = await keyboardInputService.GetKeyboardKey();
         if (IsKeyUp(key))
            index = MyMath.Repeat(index - 1, 0, _itemsValues.Count - 1);
         else if (IsKeyDown(key))
            index = MyMath.Repeat(index + 1, 0, _itemsValues.Count - 1);
         else if (key.IsEnter())
            return Option<(int, string)>.Ok((index, _itemsValues[index].Get()));
         else if (key.IsEsc() || key.IsBackspace()) return Option<(int, string)>.Ok((-1, string.Empty));
      }

      return Option<(int, string)>.Fail();
   }

   private static bool IsKeyUp(KeyboardKey key) {
      return key.KeyCode is 38 or 87 or 33; // ArrowUp 38, W 87, PageUp 33;
   }

   private static bool IsKeyDown(KeyboardKey key) {
      return key.KeyCode is 40 or 83 or 34; // ArrowDown 40, S 83, PageDown 34;
   }
}

using System.Text;

using FlashBrowser.Services.Input;
using FlashBrowser.UI.Elements.Base;
using FlashBrowser.UI.Values;

using PupaLib.Core;

namespace FlashBrowser.UI.Elements.Terminal;

public sealed class UiTerminalInput<T>(
   IKeyboardInputService keyboardInputService) :
   UiElement, IUiInput<T> {
   private readonly List<Func<T, bool>> _conditionCallbacks = [];
   private IValue<string> _promptValue = new ConstValue<string>(string.Empty);
   private IValue<(int min, int max)> _rangeValue = new ConstValue<(int min, int max)>((0, int.MaxValue));
   private IValue<char>? _secretSymbolValue;
   private ValueRef<T>? _valueRef;

   public override async Task<Option> Run(CancellationToken cancellationToken) {
      while (cancellationToken.IsCancellationRequested == false) {
         var resultOption = await Invoke(cancellationToken);
         Console.CursorVisible = false;
         if (resultOption.Out(out var result) == false) return Option.Fail();

         if (_conditionCallbacks.Count == 0 || _conditionCallbacks.All(x => x.Invoke(result))) {
            _valueRef?.Set(result);
            return Option.Ok();
         }
      }

      return Option.Ok();
   }

   public IUiInput<T> Ref(in ValueRef<T> valueRef) {
      _valueRef = valueRef;
      return this;
   }

   public IUiInput<T> WithPrompt(IValue<string> value) {
      _promptValue = value;
      return this;
   }

   public IUiInput<T> WithRange(IValue<(int min, int max)> value) {
      if (value.Get().min < 0)
         throw new Exception("Min < 0");
      _rangeValue = value;
      return this;
   }

   public IUiInput<T> WithSecret(IValue<char> value) {
      _secretSymbolValue = value;
      return this;
   }

   public IUiInput<T> WithCustomCondition(Func<T, bool> conditionCallback) {
      _conditionCallbacks.Add(conditionCallback);
      return this;
   }

   private async Task<Option<T>> Invoke(CancellationToken cancellationToken) {
      var sb = new StringBuilder();
      sb.Clear();
      Console.CursorVisible = true;
      Console.Write($"[{_promptValue.Get()}] > ");
      while (cancellationToken.IsCancellationRequested == false) {
         var cursor = Console.GetCursorPosition();
         var keyboardKey = await keyboardInputService.GetKeyboardKey();
         if (keyboardKey.IsEnter()) //Enter
         {
            if (sb.Length <= _rangeValue.Get().min)
               continue;
            try {
               var result = (T)Convert.ChangeType(sb.ToString(), typeof(T));
               Console.Write('\n');
               return Option<T>.Ok(result);
            } catch {
               continue;
            }
         }

         if (keyboardKey.IsEsc()) //Escape
         {
            Console.Write('\n');
            break;
         }

         if (keyboardKey.IsBackspace()) //Backspace
         {
            if (sb.Length < 1)
               continue;
            sb.Remove(sb.Length - 1, 1);
            Console.SetCursorPosition(cursor.Left - 1, cursor.Top);
            Console.Write(' ');
            Console.SetCursorPosition(cursor.Left - 1, cursor.Top);
            continue;
         }

         if (sb.Length >= _rangeValue.Get().max) continue;
         sb.Append(keyboardKey.Symbol);
         if (_secretSymbolValue != null) {
            Console.Write(_secretSymbolValue.Get());
            continue;
         }

         Console.Write(keyboardKey.Symbol);
      }

      return Option<T>.Fail();
   }
}

            _valueRef?.Set(result.Content);
            return Option.Ok();
         }

         break;
      }

      return Option.Fail();
   }


   public IUiVerify WithY(IValue<string> value) {
      _yValue = value;
      return this;
   }

   public IUiVerify WithN(IValue<string> value) {

      return this;
   }

      _valueRef = valueRef;
      return this;

      while (cancellationToken.IsCancellationRequested == false) {

         if (key.KeyCode == _keyYValue.Get().KeyCode) // Y 89 
            return Option<bool>.Ok(true);
         if (key.KeyCode == _keyNValue.Get().KeyCode) // N 78
            return Option<bool>.Ok(false);
      }

      return Option<bool>.Fail();
   }

   }

   public byte G { get; }
